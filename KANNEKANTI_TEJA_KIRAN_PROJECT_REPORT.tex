\section*{Executive Summary}
This report details the design, implementation, and deployment of a modern, highly scalable back-end infrastructure built for high-traffic E-Commerce platforms. Developed by \textbf{KANNEKANTI TEJA KIRAN} for submission to \textbf{ZAALIMA DEVELOPMENT PVT LTD}, the architecture transitions monolithic business processes into 6 independently deployable Spring Boot microservices (\texttt{API Gateway}, \texttt{Service Registry}, \texttt{Order Processing}, \texttt{Inventory Management}, \texttt{Payment Processing}, and \texttt{Notification Service}).

To maintain data consistency across distributed relational (\textbf{PostgreSQL}) and document (\textbf{MongoDB}) databases without two-phase commit (2PC) distributed locks, the system implements a \textbf{Choreography-based Saga Design Pattern} over \textbf{Apache Kafka}. The entire platform features resilience controls (\textbf{Resilience4j}), distributed tracing (\textbf{Zipkin} and \textbf{Micrometer}), containerization (\textbf{Docker Compose}), and orchestration manifests (\textbf{Kubernetes}).

\section{System Architecture \& Saga Design Pattern}

\subsection{Architectural Overview}
The system architecture decouples domain microservices through event-driven pub/sub channels. Client HTTP requests land on the \textbf{API Gateway} (\texttt{port 8085}), which performs JWT token validation before routing requests via Eureka service registration to the appropriate microservice.

\begin{figure}[htbp]
    \centering
    \includegraphics[width=0.95\textwidth]{architecture_diagram_output_1790789659184.jpg}
    \caption{Distributed Event-Driven Microservices Platform Architecture}
    \label{fig:architecture}
\end{figure}

\subsection{Choreography Saga Execution Flow}
Distributed transactions follow an asynchronous event choreography model across Kafka topics:
\begin{enumerate}[leftmargin=*]
    \item \textbf{Order Creation}: Client invokes \texttt{POST /api/v1/orders}. \texttt{Order Service} persists order with status \texttt{PENDING} in PostgreSQL \texttt{order\_db} and publishes \texttt{OrderCreatedEvent} to Kafka topic \texttt{order-events}.
    \item \textbf{Inventory Stock Reservation}: \texttt{Inventory Service} consumes \texttt{OrderCreatedEvent}, deducts available stock in \texttt{inventory\_db}, and emits \texttt{INVENTORY\_RESERVED}.
    \item \textbf{Payment Processing}: \texttt{Payment Service} consumes \texttt{OrderCreatedEvent}, processes customer charge in \texttt{payment\_db}, and emits \texttt{PAYMENT\_PROCESSED}.
    \item \textbf{Saga State Completion}: \texttt{Order Service} listens to Kafka event topics. When both \texttt{inventoryReserved = true} and \texttt{paymentProcessed = true}, it updates order status to \textbf{\texttt{CONFIRMED}} and emits \texttt{ORDER\_COMPLETED}.
    \item \textbf{Compensating Transactions (Rollback)}: If stock reservation or payment fails, an \texttt{ORDER\_CANCELLED} event is published to trigger automated refund and inventory release rollbacks.
    \item \textbf{Notification Alert}: \texttt{Notification Service} listens for completed/cancelled events and logs customer SMS/email alerts in MongoDB \texttt{notification\_db}.
\end{enumerate}

\section{Microservices Implementation Code}

\subsection{Order Controller Implementation}
\begin{lstlisting}[language=Java, caption=OrderController.java - Order Creation \& Event Publishing]
@RestController
@RequestMapping("/api/v1/orders")
public class OrderController {
    private final OrderRepository orderRepository;
    private final KafkaProducer kafkaProducer;

    public OrderController(OrderRepository orderRepository, KafkaProducer kafkaProducer) {
        this.orderRepository = orderRepository;
        this.kafkaProducer = kafkaProducer;
    }

    @PostMapping
    @CircuitBreaker(name = "inventoryServiceCB", fallbackMethod = "orderFallback")
    public ResponseEntity<Order> createOrder(@RequestBody Order orderRequest) {
        orderRequest.setOrderNumber("ORD-" + UUID.randomUUID().toString().substring(0, 8));
        orderRequest.setStatus(OrderStatus.PENDING);
        Order savedOrder = orderRepository.save(orderRequest);

        String eventPayload = String.format(
            "{\"eventType\":\"ORDER_CREATED\", \"orderNumber\":\"%s\", \"productId\":\"%s\", \"quantity\":%d, \"totalAmount\":%s, \"customerId\":\"%s\"}",
            savedOrder.getOrderNumber(), savedOrder.getProductId(), savedOrder.getQuantity(),
            savedOrder.getTotalAmount(), savedOrder.getCustomerId()
        );
        kafkaProducer.sendEvent("order-events", savedOrder.getOrderNumber(), eventPayload);
        return ResponseEntity.ok(savedOrder);
    }
}
\end{lstlisting}

\subsection{Multi-Stage Dockerfile Template}
\begin{lstlisting}[language=bash, caption=Multi-Stage Dockerfile for Spring Boot Microservices]
# Stage 1: Build JAR using Maven
FROM maven:3.9.6-eclipse-temurin-17-alpine AS build
WORKDIR /app
COPY pom.xml .
COPY src ./src
RUN mvn clean package -DskipTests

# Stage 2: Runtime JRE Container
FROM eclipse-temurin:17-jre-alpine
WORKDIR /app
COPY --from=build /app/target/*.jar app.jar
EXPOSE 8081
ENTRYPOINT ["java", "-jar", "app.jar"]
\end{lstlisting}
